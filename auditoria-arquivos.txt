#!/bin/bash
GREEN='\033[0;32m'
CYAN='\033[0;36m'
YELLOW='\033[1;33m'
NC='\033[0m'

echo -e "${CYAN}🚀 Iniciando Ambiente IA no NVMe Gigante (/workspace)...${NC}"

# 1. Tenta subir os containers se já existirem, senão cria-os
if docker start ollama-service open-webui-gui 2>/dev/null; then
    echo -e "${GREEN}📦 Containers iniciados com sucesso!${NC}"
else
    echo -e "${YELLOW}📦 Criando novos containers no /workspace...${NC}"
    docker run -d \
      --name ollama-service \
      --restart always \
      -v /workspace/ollama-models:/root/.ollama \
      -p 11434:11434 \
      ollama/ollama:latest

    docker run -d \
      -p 8080:8080 \
      --add-host=host.docker.internal:host-gateway \
      -v "/workspace/AI-data/open-webui:/app/backend/data" \
      --name open-webui-gui \
      --restart always \
      ghcr.io/open-webui/open-webui:main
fi

# 2. Aguarda o serviço do Ollama estar pronto para receber comandos
echo -ne "${YELLOW}⏳ Aguardando API do Ollama... ${NC}"
while ! docker exec ollama-service ollama list > /dev/null 2>&1; do
    echo -ne "."
    sleep 1
done
echo -e "${GREEN} OK!${NC}"

# 3. AQUECIMENTO E SAUDAÇÃO (Carrega o Mistral na RAM)
echo -e "${CYAN}🧠 Aquecendo o Mistral NeMo (7.1 GB)...${NC}"
SAUDACAO=$(docker exec ollama-service ollama run mistral-nemo "Responda apenas: 'Aguardando instruções, Carlos! Sistema pronto no NVMe.'")

echo -e "\n${GREEN}▶ $SAUDACAO${NC}"

# Alias temporário para a sessão
alias ollama='docker exec -it ollama-service ollama'

echo -e "${GREEN}✅ Tudo pronto para o trabalho!${NC}"
#!/bin/bash
RED='\033[0;31m'
GREEN='\033[0;32m'
CYAN='\033[0;36m'
YELLOW='\033[1;33m'
NC='\033[0m'

echo -e "${CYAN}⏹️  Encerrando Laboratório de IA...${NC}"

# 1. Descarrega os modelos da RAM (Keep-alive 0)
echo -e "${YELLOW}🧠 Descarregando Mistral e Command-R da RAM...${NC}"
docker exec ollama-service ollama stop mistral-nemo 2>/dev/null
docker exec ollama-service ollama stop command-r 2>/dev/null

# 2. Para os containers
echo -e "${YELLOW}📦 Parando containers...${NC}"
docker stop open-webui-gui ollama-service > /dev/null 2>&1

# 3. Sincronização e Limpeza de Cache (Crucial para 32GB RAM)
echo -e "${CYAN}🧹 Sincronizando NVMe e liberando memória...${NC}"
sudo sync && echo 3 | sudo tee /proc/sys/vm/drop_caches > /dev/null

# 4. Verificação final
MEM_FREE=$(free -h | grep Mem | awk '{print $4}')
echo -e "${GREEN}✅ Sucesso! Memória RAM disponível: $MEM_FREE${NC}"
echo -e "${GREEN}🚀 Hardware liberado, Carlos!${NC}"
#!/bin/bash

# Cores
CYAN='\033[0;36m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
RED='\033[0;31m'
NC='\033[0m'

echo -e "${CYAN}==================================================${NC}"
echo -e "${CYAN}         📊 STATUS DO ECOSSISTEMA IA (NVMe)       ${NC}"
echo -e "${CYAN}==================================================${NC}"

# 1. Modelo Ativo
echo -e "\n${YELLOW}🧠 [MODELO EM EXECUÇÃO]${NC}"
# Filtra a saída do ps para pegar Nome e Tamanho
MODELO_ATIVO=$(docker exec ollama-service ollama ps 2>/dev/null | grep -v "NAME" | awk '{print $1 " (" $3 ")"}')

if [ -n "$MODELO_ATIVO" ]; then
    echo -e "${GREEN}▶ $MODELO_ATIVO${NC}"
else
    # Se não houver modelo fixo na RAM, mostra o uso atual de CPU do container
    CPU_USAGE=$(docker stats --no-stream ollama-service --format "{{.CPUPerc}}")
    echo -e "${RED}💤 Standby (Ollama CPU: $CPU_USAGE)${NC}"
fi

# 2. Containers
echo -e "\n${YELLOW}📦 [CONTAINERS]${NC}"
docker ps --format "table {{.Names}}\t{{.Status}}\t{{.Ports}}" | grep -E "Names|open-webui|ollama"

# 3. Armazenamento
echo -e "\n${YELLOW}💾 [ARMAZENAMENTO - NVMe /WORKSPACE]${NC}"
CAMINHO_LOCAL="/workspace/AI-data/open-webui"
if [ -d "$CAMINHO_LOCAL" ]; then
    TAMANHO_PASTA=$(du -sh "$CAMINHO_LOCAL" | cut -f1)
    TAMANHO_DB=$(du -sh "$CAMINHO_LOCAL/webui.db" 2>/dev/null | cut -f1)
    echo -e "${GREEN}✅ Pasta de Dados: ${YELLOW}$TAMANHO_PASTA${NC}"
    echo -e "${GREEN}✅ Base de Dados (webui.db): ${YELLOW}${TAMANHO_DB:-0}${NC}"
fi

# 4. Memória RAM
echo -e "\n${YELLOW}🧠 [MEMÓRIA RAM]${NC}"
(
  echo "DESCRIÇÃO TOTAL USADA LIVRE CACHE DISPONÍVEL"
  free -h | grep "Mem:" | awk '{print "Memória-Física " $2 " " $3 " " $4 " " $6 " " $7}'
  free -h | grep "Swap:" | awk '{print "Swap-Disco " $2 " " $3 " " $4 " - -"}'
) | column -t

echo -e "\n${CYAN}==================================================${NC}"
